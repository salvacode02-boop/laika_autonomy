cmake_minimum_required(VERSION 3.5)
project(laika_imu_relay)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
if(NOT CMAKE_CXX_STANDARD)
  set(CMAKE_CXX_STANDARD 14)
endif()
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(unitree_go REQUIRED)
add_executable(lowstate_imu_relay src/lowstate_imu_relay.cpp)
ament_target_dependencies(lowstate_imu_relay rclcpp sensor_msgs unitree_go)
install(TARGETS lowstate_imu_relay DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
ament_package()
